import numpy as np

# Task 1: Element-wise Mathematical Functions
arr = np.array([1, 2, 3, 4])
exp_res = np.exp(arr)
print("Task 1 Output:")
print(exp_res)

# Task 2: Finding the Index of Maximum Value
probs = np.array([0.1, 0.7, 0.2])
pred_class = probs.argmax()
print("\nTask 2 Output:")
print("Predicted Class Index:", pred_class)

# Task 3: Element-wise vs Dot Product Comparison
a = np.array([1, 2])
b = np.array([3, 4])
elem_mult = a * b
dot_prod = a @ b
print("\nTask 3 Output:")
print("Element-wise multiply (a * b):", elem_mult)
print("Dot product (a @ b):", dot_prod)

# Task 4: Simulating Linear Regression Predictions
X = np.array([[2, 1],
              [4, 3],
              [6, 5]])
w = np.array([0.5, 2.0])
preds = X @ w
print("\nTask 4 Output:")
print("Predictions (X @ w):", preds)

# Task 5: Weighted Sum Equivalence
weights = np.array([0.2, 0.5, 0.3])
values = np.array([80, 90, 70])
sum_elem = (weights * values).sum()
dot_elem = weights @ values
print("\nTask 5 Output:")
print("sum(weights * values):", sum_elem)
print("weights @ values:", dot_elem)
print("Are both identical?:", np.isclose(sum_elem, dot_elem))
