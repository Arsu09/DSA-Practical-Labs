TASK 3: Diagnosing and Fixing Shape Mismatches
------------------------------------------------------------------------------
[PYTHON CODE]
import numpy as np

arr1 = np.ones((3, 4), dtype=int)
arr2 = np.array([10, 20, 30])

print("Attempting operation with shapes:", arr1.shape, "and", arr2.shape)
try:
    bad_sum = arr1 + arr2
except ValueError as err:
    print("Captured Expected Error:", err)

# Fixing shape using reshape(-1, 1) to treat as column vector (3, 1)

fixed_result = arr1 + arr2_fixed

print("\nFixed Shape of second array:", arr2_fixed.shape)

import numpy as np

np.random.seed(101)
raw_features = np.random.normal(loc=50, scale=15, size=(10, 3))

# Single-line vectorised feature standardisation
standardised_features = (raw_features - raw_features.mean(axis=0)) / raw_features.std(axis=0)

print("Raw Feature Matrix (10x3):\n", np.round(raw_features, 2))
