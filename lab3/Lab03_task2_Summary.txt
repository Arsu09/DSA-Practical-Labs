# Create tasks.py in lab3
Set-Content -Path "lab3\tasks.py" -Value @"
import pandas as pd
import numpy as np

# Task 1: Identify and Remove Duplicate Rows
df1 = pd.DataFrame({
    'id': [1, 2, 1, 3],
    'name': ['Ali', 'Sara', 'Ali', 'Omar']
})
dup_count = df1.duplicated().sum()
df1_clean = df1.drop_duplicates()
print("Task 1 Output:")
print("Duplicate count:", dup_count)
print("Clean DataFrame:\n", df1_clean)

# Task 2: Convert Text-Based Numbers to Numeric
df2 = pd.DataFrame({'price': ['10.5', '20.0', 'missing', '15.2']})
df2['price'] = pd.to_numeric(df2['price'], errors='coerce')
print("\nTask 2 Output:")
print("dtypes:\n", df2.dtypes)
print("DataFrame:\n", df2)

# Task 3: Standardise Categorical Text Data
df3 = pd.DataFrame({'name': [' Ali ', 'ali', 'ALI', 'Sara']})
df3['name'] = df3['name'].str.strip().str.lower()
print("\nTask 3 Output:")
print(df3)

# Task 4: Impute Missing Values for Model Training
train = pd.DataFrame({'age': [22, np.nan, 35, 19, 40]})
test = pd.DataFrame({'age': [np.nan, 28, 30]})
train_median = train['age'].median()
train['age'] = train['age'].fillna(train_median)
test['age'] = test['age'].fillna(train_median)
print("\nTask 4 Output:")
print("Calculated Train Median:", train_median)
print("Train DataFrame:\n", train)
print("Test DataFrame:\n", test)

# Task 5: Analytical Handling of Missing Data
df5 = pd.DataFrame({
    'ID': [101, np.nan, 103, 104, 105],
    'Score': [88, 72, np.nan, 91, 60]
})
df5_clean = df5.dropna(subset=['ID']).copy()
df5_clean['Score'] = df5_clean['Score'].fillna(df5_clean['Score'].mean())
print("\nTask 5 Output:")
print(df5_clean)
"@
