# Task 2: Filtering Data with List Comprehensions
raw_readings = [12.5, 45.0, 102.1, 8.2, 150.3, 33.7]
filtered_readings = [r for r in raw_readings if r > 40.0]
print("\nTask 2 Output:")
print(filtered_readings)

# Task 3: Transforming Data with Dictionary Comprehensions
user_ids = [101, 102, 103]
user_map = {uid: f"User_{uid}" for uid in user_ids}
print("\nTask 3 Output:")
print(user_map)

# Task 4: Handling Dirty Data Gracefully
def safe_convert(val):
    try:
        return int(val)
    except (ValueError, TypeError):
        return 'Invalid'

print("\nTask 4 Output:")
print("safe_convert('10'):", safe_convert('10'))
print("safe_convert('N/A'):", safe_convert('N/A'))

# Task 5: Processing a Table of Records
data = [{'val': '10'}, {'val': '20'}, {'val': 'error'}, {'val': '30'}]
results = [safe_convert(item['val']) for item in data]
print("\nTask 5 Output:")
print(results)
