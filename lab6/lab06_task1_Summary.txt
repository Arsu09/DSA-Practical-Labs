import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
from sklearn.pipeline import make_pipeline
from sklearn.preprocessing import StandardScaler
import os
import subprocess

def print_sys_info():
    print("--- Terminal Output ---")
    try:
        print("whoami:", subprocess.getoutput("whoami"))
        print("hostname:", subprocess.getoutput("hostname"))
        print("pwd:", os.getcwd())
    except:
        pass

print("=== TASK 1: Implement Reproducible Data Splitting ===")
X = np.array([[1], [2], [3], [4]])
y = np.array([0, 1, 0, 1])
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.5, random_state=42, stratify=y)
print("X_train shape:", X_train.shape)
print("X_test shape:", X_test.shape)
print_sys_info()

print("\n=== TASK 2: Basic Model Fit and Predict Cycle ===")
X_train2 = np.array([[1], [2], [3]])
y_train2 = np.array([2, 4, 6])
X_test2 = np.array([[4], [5]])
model = LinearRegression()
model.fit(X_train2, y_train2)
print("Predictions for X_test:", model.predict(X_test2))
print_sys_info()

print("\n=== TASK 3: Construct a Leak-Free Pipeline ===")
X_train3 = np.array([[10], [20], [30]])
y_train3 = np.array([100, 200, 300])
X_test3 = np.array([[40], [50]])
pipe = make_pipeline(StandardScaler(), LinearRegression())
pipe.fit(X_train3, y_train3)
print("Pipeline Predictions for test set:", pipe.predict(X_test3))
print_sys_info()

print("\n=== TASK 4: Model Coefficient Inspection ===")
X_train4 = np.array([[1, 2], [2, 4], [3, 6]])
y_train4 = np.array([5, 10, 15]) 
model4 = LinearRegression().fit(X_train4, y_train4)
print("Model Coefficients:", model4.coef_)
print("Explanation: These coefficients represent the 'weight' or importance of each feature.")
print_sys_info()
