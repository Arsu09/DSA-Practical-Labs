matplotlib.use('Agg')
import matplotlib.pyplot as plt
import seaborn as sns
import pandas as pd
import numpy as np

print("=== TASK 1: Compute a Correlation Matrix ===")
numeric_df = df.select_dtypes(include=[np.number])
corr_matrix = numeric_df.corr().round(2)
print(corr_matrix)

print("\n=== TASK 2: Identify Multicollinearity ===")
print("High feature-to-feature correlation pair: total_bill and size (r = 0.60)")

print("\n=== TASK 3: Linear vs Non-Linear Relationship Analysis ===")
np.random.seed(42)
x = np.linspace(-5, 5, 100)
y = x**2
r = np.corrcoef(x, y)[0, 1]
print(f"Correlation coefficient r for U-shaped relationship: {r:.4f}")

fig, ax = plt.subplots(figsize=(6, 4))
ax.scatter(x, y, color='purple', alpha=0.7)
ax.set_title('U-Shaped Non-Linear Relationship (r ≈ 0)')
ax.set_xlabel('x')
ax.set_ylabel('y = x²')
fig.savefig('task3_nonlinear.png', dpi=120, bbox_inches='tight')
plt.close(fig)

print("\n=== TASK 4: Categorical Comparison via Box Plot ===")
fig, ax = plt.subplots(figsize=(6, 4))
sns.boxplot(data=df, x='smoker', y='tip', ax=ax)
ax.set_title('Tip Amount by Smoker Status')
fig.savefig('task4_boxplot.png', dpi=120, bbox_inches='tight')
plt.close(fig)

print("\n=== TASK 5: Confounder Hypothesis Testing ===")
fig, ax = plt.subplots(figsize=(6, 4))
sns.boxplot(data=df, x='smoker', y='total_bill', ax=ax)
ax.set_title('Total Bill by Smoker Status (Potential Confounder)')
fig.savefig('task5_confounder.png', dpi=120, bbox_inches='tight')
